### Contrast script

```js
function hexToRgb(h){h=h.replace('#','');const n=parseInt(h,16);return [(n>>16)&255,(n>>8)&255,n&255];}
function lum([r,g,b]){const f=c=>{const s=c/255;return s<=0.03928?s/12.92:Math.pow((s+0.055)/1.055,2.4);};return 0.2126*f(r)+0.7152*f(g)+0.0722*f(b);}
function cr(a,b){const A=lum(hexToRgb(a)),B=lum(hexToRgb(b));const l=Math.max(A,B),d=Math.min(A,B);return (l+0.05)/(d+0.05);}
function comp(fg,al,bg){const f=hexToRgb(fg),b=hexToRgb(bg);const m=f.map((x,i)=>Math.round(al*x+(1-al)*b[i]));return '#'+m.map(x=>x.toString(16).padStart(2,'0')).join('');}
// every row above is cr(fg, bg) over the hex pairs shown; composites use comp() first
```
